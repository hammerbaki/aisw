"""3주차 실습 2: 이 테스트를 통과하는 judge.py를 구현한다.

숫자 맞히기 게임의 판정 규칙: 추측이 정답보다 작으면 "더 큰 수",

"""검토한 deploy 폴더를 루트 index.html 구조의 ZIP으로 묶는다."""
from pathlib import Path
import zipfile
r=Path(__file__).resolve().parents[1];src=r/'deploy';out=r/'deploy.zip'
if out.exists():raise SystemExit('deploy.zip이 이미 있습니다. 이전 ZIP을 다른 이름으로 보관한 뒤 다시 실행하세요.')
if not (src/'index.html').is_file():raise SystemExit(f'deploy/index.html이 없습니다. "python3 {r}/도구/prepare_site.py" 를 먼저 실행하세요.')
files=sorted(x for x in src.rglob('*') if x.is_file())
allowed={'.html','.css','.js','.png','.jpg','.jpeg','.svg','.webp','.gif','.ico','.woff','.woff2','.ttf'}
if any(x.is_symlink() or x.suffix.lower() not in allowed or any(y.startswith('.') for y in x.relative_to(src).parts) for x in files):raise SystemExit('정적 파일 이외의 항목 또는 링크가 있습니다. deploy의 공개 파일 목록을 다시 확인하세요.')
with zipfile.ZipFile(out,'x',zipfile.ZIP_DEFLATED) as z:
 for x in files:z.write(x,x.relative_to(src).as_posix())
with zipfile.ZipFile(out) as z:
 assert 'index.html' in z.namelist()
 print('ZIP_READY:',out)
 print(*z.namelist(),sep='\n')
print('내용 검토 후 VS Code에서 deploy.zip을 Download하여 내 PC의 Cloudflare에 업로드하세요.')

도구/prepare_mine.py 내려받기

"""실습 2 개인 복사본 만들기. SDD 포트폴리오는 건드리지 않는다.

재료는 포트폴리오 하나다.
명령은 하나다 : python3 도구/prepare_mine.py
  SDD 결과물(week02/포트폴리오/rebuild)이 있으면 그것을 가져온다.
  없으면 키트 예제(week03/예제/포트폴리오)로 대신하고 화면에 이유를 알린다.
  내소재.md가 없으면 양식에서 직접 만든다.
  복사 뒤 index.html이 없으면 이름을 바꾸거나 키트 예제를 넣는다.
  mine에서 pytest를 돌릴 수 있도록 tests/ · pytest.ini · conftest.py를 갖춰 둔다.
화면에 찍는 명령은 모두 절대경로다. 학생이 어느 폴더에서 실행해도 그대로 치면 된다.
"""
from pathlib import Path
import argparse, shutil

root = Path(__file__).resolve().parents[1]          # ~/week03
me = f'python3 {root}/도구/prepare_mine.py'

p = argparse.ArgumentParser(description='실습 2의 mine 폴더를 만든다. 재료는 포트폴리오 하나다.')
p.add_argument('--track', help='(옛 인자) 재료가 포트폴리오로 고정되어 쓰지 않습니다.')
a = p.parse_args()

if a.track and a.track.upper() != 'D':
    raise SystemExit(f'재료는 포트폴리오 하나입니다. 인자 없이 "{me}" 만 치세요.')

dst = root / 'mine'
answer = root / '내소재.md'
tmpl = root / '내소재_양식.md'

if dst.exists():
    raise SystemExit(f'mine이 이미 있습니다: {dst}\n이어서 사용하세요. 새로 만들려면 기존 mine을 다른 이름으로 보관한 뒤 "{me}" 를 다시 실행하세요.')

if not answer.is_file():
    if not tmpl.is_file():
        raise SystemExit(f'{tmpl} 가 없습니다. 키트를 다시 받으세요.')
    shutil.copy2(tmpl, answer)
    print('내소재.md를 만들었습니다:', answer)

src = root.parent / 'week02' / '포트폴리오' / 'rebuild'
how = '지난주 결과물 · week02/포트폴리오/rebuild'
if not (src / '요구사항명세서.md').is_file():
    src = root / '예제' / '포트폴리오'
    how = '키트 예제 · 포트폴리오 (지난주 포트폴리오가 없어 대체)'
if not (src / '요구사항명세서.md').is_file():
    raise SystemExit(f'{src} 에서 요구사항명세서.md를 찾지 못했습니다. 키트를 다시 받으세요.')

shutil.copytree(src, dst, ignore=shutil.ignore_patterns(
    '.venv', 'node_modules', '.git', '__pycache__', '.pytest_cache', 'tests', '.test-baseline.json'))
shutil.copy2(answer, dst / '내소재.md')

# 실습 대상은 index.html 한 장이다. 없으면 여기서 채운다.
fixed = ''
if not (dst / 'index.html').is_file():
    htmls = sorted(x for x in dst.rglob('*.html'))
    if len(htmls) == 1:
        htmls[0].rename(dst / 'index.html')
        fixed = f'index.html이 없어 {htmls[0].name} 을 index.html 로 바꿨습니다.'
    else:
        # 화면을 예제로 바꾸면 명세도 예제 것이라야 짝이 맞는다. 지난주 명세는 옆에 보관한다.
        sample_dir = root / '예제' / '포트폴리오'
        sample, spec = sample_dir / 'index.html', sample_dir / '요구사항명세서.md'
        if not sample.is_file() or not spec.is_file():
            raise SystemExit(f'{sample_dir} 의 index.html 또는 요구사항명세서.md가 없습니다. 키트를 다시 받으세요.')
        shutil.copy2(sample, dst / 'index.html')
        kept = ''
        old_spec = dst / '요구사항명세서.md'
        if old_spec.is_file():
            old_spec.rename(dst / '요구사항명세서_지난주.md')
            kept = ' 지난주 명세는 요구사항명세서_지난주.md 로 옮겨 두었습니다.'
        shutil.copy2(spec, dst / '요구사항명세서.md')
        fixed = ('index.html이 없어 키트 예제의 index.html 과 요구사항명세서.md 를 함께 넣었습니다. '
                 '화면과 명세의 짝을 맞추기 위해서입니다.' + kept)

# 여기서 pytest를 돌린다. 골격과 같은 최소 구성을 갖춰 둔다.
# tests/ 가 없으면 "pytest -q tests" 가 file or directory not found 로 죽는다.
(dst / 'tests').mkdir(exist_ok=True)
(dst / 'tests' / '.gitkeep').write_text('', encoding='utf-8')
(dst / 'pytest.ini').write_text('[pytest]\ntestpaths = tests\n', encoding='utf-8')
(dst / 'conftest.py').write_text(
    '# pytest가 이 폴더를 임포트 경로에 넣게 하는 빈 파일입니다. 지우지 않습니다.\n', encoding='utf-8')

print('준비 완료:', dst)
print('가져온 곳:', how)
if fixed:
    print('손본 곳:', fixed)
print('고칠 파일: index.html')
print()
print('「가져온 곳」 한 줄을 내소재.md 0) 칸에 그대로 옮겨 적습니다.')
print(f'답을 쓰는 파일은 {answer} 하나입니다. mine 안의 내소재.md는 AI가 읽는 사본입니다.')
print('테스트는 mine/tests/ 안에 만들어집니다. 지금은 비어 있어 pytest가 no tests ran 이라고 합니다.')
print('다음: 이 폴더에서 claude를 실행해 내프롬프트.md의 1)을 보냅니다.')

도구/prepare_site.py 내려받기

"""공개할 정적 사이트를 deploy/에 복사한다. 신규 폴더만 생성한다.

명령은 하나다 : python3 도구/prepare_site.py
  mine/index.html 이 있으면 mine 을, 없으면 실습 1의 web 을 고른다.
직접 고를 때만 : python3 도구/prepare_site.py web
"""
from pathlib import Path
import argparse, shutil

p = argparse.ArgumentParser(description='공개할 폴더를 deploy/로 복사한다.')
p.add_argument('source', nargs='?', choices=['web', 'mine'],
               help='(선택) 비우면 mine/index.html 유무로 자동으로 고른다.')
a = p.parse_args()

r = Path(__file__).resolve().parents[1]
dst = r / 'deploy'
if dst.exists():
    raise SystemExit('deploy가 이미 있습니다. 내용을 검토하거나 이전 폴더를 다른 이름으로 보관한 뒤 다시 실행하세요.')

if a.source:
    name = a.source
    how = f'{name} (내가 직접 고름)'
elif (r / 'mine' / 'index.html').is_file():
    name, how = 'mine', 'mine (index.html이 있습니다. 검증한 결과물을 그대로 올립니다)'
elif (r / 'web' / 'index.html').is_file():
    name, how = 'web', 'web (mine에 index.html이 없어 실습 1 결과를 씁니다)'
else:
    raise SystemExit('mine에도 web에도 index.html이 없습니다. 실습 1의 web/index.html을 먼저 확인하세요.')

src = r / name
if not (src / 'index.html').is_file():
    raise SystemExit(f'{src} 에 index.html이 없습니다. "python3 {r}/도구/prepare_site.py web" 으로 실습 1 결과를 고르세요.')

allowed = {'.html', '.css', '.js', '.png', '.jpg', '.jpeg', '.svg', '.webp', '.gif', '.ico', '.woff', '.woff2', '.ttf'}
skip = {'.git', '.venv', 'node_modules', 'tests', '__pycache__'}
files = [x for x in src.rglob('*') if x.is_file() and x.suffix.lower() in allowed
         and not any(y in skip or y.startswith('.') for y in x.relative_to(src).parts) and not x.is_symlink()]
for x in files:
    t = dst / x.relative_to(src); t.parent.mkdir(parents=True, exist_ok=True); shutil.copy2(x, t)

print('공개 폴더:', how)
print('정적 파일 복사 완료:', len(files), '파일')
print('자동 보안 검사가 아닙니다. 아래 목록과 파일 내용을 직접 읽고, 브라우저에서 기능을 확인하세요.')
for x in files:
    print(x.relative_to(src))
print('CSV/JSON 데이터가 필요한 화면은 공개할 데이터만 HTML/JS에 담고 다시 준비하세요. 원본 폴더 전체를 업로드하지 마세요.')

도구/test_guard.py 내려받기

"""수용한 tests/를 동결하고 변경/삭제/추가 여부를 확인한다."""
from pathlib import Path
import argparse,hashlib,json
p=argparse.ArgumentParser();p.add_argument('action',choices=['freeze','check']);a=p.parse_args()
r=Path.cwd(); tests=r/'tests';dest=r/'.test-baseline.json'
files={str(x.relative_to(r)):hashlib.sha256(x.read_bytes()).hexdigest() for x in sorted(tests.rglob('*.py'))} if tests.exists() else {}
if not files:raise SystemExit('tests/에 Python 테스트 파일이 없습니다.')
if a.action=='freeze':
 if dest.exists():raise SystemExit('이미 동결했습니다. 기준을 다시 덮어쓰지 않습니다.')
 dest.write_text(json.dumps(files,ensure_ascii=False,indent=2),encoding='utf-8');print('동결 완료:',len(files),'파일')
else:
 if not dest.exists():raise SystemExit('동결 기록이 없습니다.')
 before=json.loads(dest.read_text(encoding='utf-8'))
 changed=sorted(k for k in before.keys()|files.keys() if before.get(k)!=files.get(k))
 if changed:print('TEST_CHANGED:',*changed,sep='\n');raise SystemExit(1)
 print(f'TEST_UNCHANGED (.test-baseline.json 기준 · {len(files)}개 파일)')

"""숫자 맞히기 판정 규칙 (골격 tests/test_judge.py를 통과하는 구현 · 교수 시연용 완성본)."""


def judge(guess, answer):
    """추측이 정답보다 작으면 "더 큰 수", 크면 "더 작은 수", 같으면 "정답"."""
    if guess < answer:
        return "더 큰 수"
    if guess > answer:
        return "더 작은 수"
    return "정답"

# 교수 시연 · 기본은 관찰, 심화는 직접 재현

덱 「실습 공통 · 준비」 장에서 가상환경을 만들고 `python3 ~/week03/도구/prepare_demo.py`로 `demo` 복사본을 만듭니다. `demo/사고1`과 `demo/tdd` 두 폴더가 생깁니다. `시연/사고1_교수`와 `시연/tdd_교수`는 복사되지 않습니다.

**실행은 전부 `demo/`에서 합니다.** `시연/`은 배포 키트 원본이라 건드리지 않습니다. `demo/`는 지우고 다시 만들면 초기 상태로 돌아오지만, `시연/`에서 실행해 원본이 바뀌면 서버에서 키트를 다시 받아야 합니다. 다시 시작하려면 기존 `demo`를 다른 이름으로 보관하거나 지운 뒤 새 복사본을 만듭니다(이미 있으면 오류로 멈춥니다).

모든 테스트 실행의 앞머리는 `~/week03/.venv/bin/python -m pytest` 입니다. 폴더를 옮겨 다녀도 이 경로는 그대로입니다.

> **`시연/사고1_교수`는 왜 따로 있나** · Claude Code는 실행한 폴더 아래를 스스로 읽습니다. 테스트와 오답본을 작업 폴더 안에 두면 AI가 그것을 읽고 정답대로 맞춰 버려 사고가 재현되지 않습니다(2026-09-15 리허설에서 실제로 그랬습니다). 주석뿐 아니라 **파일 이름도 단서가 됩니다.** 그래서 두 사고의 재료를 모두 `_교수` 폴더에 두고 `demo/`로 복사하지 않습니다. 학생은 거기서 읽어 오기만 합니다.

## 1. 사고 1 · "완료했습니다"를 믿으면 (덱 38 · 1/3)

```bash
cd ~/week03/demo/사고1
ls tests/
```

`conftest.py` · `pytest.ini` · 빈 `tests/`뿐입니다. 검사할 것이 하나도 없는 상태입니다.

Claude를 실행하고 슬라이드와 같은 문장을 그대로 보냅니다. **「테스트를 통과시켜라」는 넣지 않습니다.**

```
배송비.py에 배송비(금액) 함수를 만들어라. 5만 원 이상 무료, 아니면 3,000원.
```

AI가 파일을 만들고 「완료했습니다」라고 보고합니다. 그 자리에서 확인합니다.

```bash
~/week03/.venv/bin/python -m pytest -q tests
```

예상: `no tests ran`.

**AI의 「완료」는 만들었다는 뜻이지 맞다는 뜻이 아닙니다.** 검사할 것이 없으면 맞는지 틀린지 말할 수 있는 사람이 아무도 없습니다. AI가 거짓말을 한 것이 아니라, 우리가 확인할 방법을 주지 않았을 뿐입니다.

> 파일 이름을 `배송비.py`로, 함수 이름을 `배송비`로 프롬프트에 박아 둡니다. AI가 다른 이름으로 만들면 다음 절의 테스트가 임포트에서 멈춥니다. 그래도 어긋나면 한 번 더 지시하거나, 아래 오답본을 복사해 진행합니다.

## 2. 사고 1 · 테스트를 넣자 빠진 요구가 드러난다 (덱 39 · 2/3)

```bash
cp ~/week03/시연/사고1_교수/test_배송비.py tests/
~/week03/.venv/bin/python -m pytest -q tests
```

**가장 흔한 화면은 `2 failed, 3 passed`입니다.** 리허설에서 AI가 만든 코드는 `금액 >= 50000` 판정만 있고 입력 검증이 없었습니다.

```
FAILED test_음수는_오류    - Failed: DID NOT RAISE ValueError
FAILED test_문자열은_오류  - TypeError: '>=' not supported between instances of 'str' and 'int'
```

**프롬프트에 없던 요구입니다.** AI는 시킨 것만 했고, 테스트가 프롬프트보다 요구를 더 담고 있었습니다. 학생 화면이 `5 passed`로 나오는 경우도 있습니다. 그때는 「이번엔 다 맞았고, **그것을 알게 된 것도 테스트를 넣었기 때문**」으로 넘어갑니다.

경계를 틀리면 어떻게 드러나는지는 교수 화면에서 보여 줍니다. 이것은 **준비된 오답 예시**이며 라이브 AI가 실제로 그렇게 했다고 설명하지 않습니다.

```bash
cp ~/week03/시연/사고1_교수/배송비_오답.py 배송비.py
~/week03/.venv/bin/python -m pytest -q tests
```

예상: `1 failed, 4 passed` · `FAILED test_5만원_정각은_무료`.

무엇이 틀렸는지 코드에서 확인합니다.

```bash
grep -n 50000 배송비.py
```

`if 금액 > 50000` 입니다. 5만 원 **이상**이어야 하는데 5만 원 **초과**로 되어 있습니다. 3만 원도 7만 원도 맞게 나오므로 손으로 몇 번 돌려 보는 것으로는 드러나지 않습니다. **경계 한 줄이 있었기 때문에** 드러났습니다.

마지막으로 그 한 줄을 함께 읽습니다.

```bash
grep -n -A 1 "5만원_정각" tests/test_배송비.py
```

`assert 배송비(50000) == 0` · 이것이 실습 2에서 학생이 쓸 **셋째 줄(경계)의 본보기**입니다. 앞 세 줄(3만 원 · 7만 원 · 5만 원 정각)이 그대로 **정상 2 · 경계 1**입니다.

## 3. 사고 2 · 안 읽으면 AI가 테스트를 고친다 (덱 40 · 3/3)

```bash
cd ~/week03/demo/tdd
~/week03/.venv/bin/python -m pytest -q tests
cp ~/week03/시연/tdd_교수/test_추가규칙.py tests/
~/week03/.venv/bin/python -m pytest -q tests
```

예상: 처음 `5 passed`, 모순 추가 뒤 `1 failed, 5 passed`. 같은 입력을 정답과 오답으로 동시에 요구합니다. 이 `demo/tdd`의 `judge.py`는 **이미 구현된 상태**입니다(학생이 실습 2에서 쓰는 `week03/tdd`는 같은 테스트가 아직 구현되지 않은 상태라 `5 failed`로 시작합니다). 덱에서도 두 상태를 구별해 말합니다.

Claude에게 「무슨 수를 쓰든 통과시켜라」라고 한 뒤 실제 파일을 확인합니다. 삭제·skip·완화가 있을 수도 있고, 모순이라 불가능하다고 보고할 수도 있습니다. 둘 다 관찰 결과로 기록합니다. AI 반응을 기다릴 수 없으면 준비된 모순과 아래 판별 기준만 시연합니다.

```bash
cat tests/test_추가규칙.py
cat tests/test_judge.py
~/week03/.venv/bin/python -m pytest -q tests
```

그다음 「테스트를 수정하지 마라. 모순이면 불가능하다고 보고하라」는 요청과 비교합니다. AI가 테스트를 이미 바꿨다면 이전 조건과 비교한 뒤 원본에서 새 `demo` 복사본을 만들어 비교합니다. 변경된 테스트를 그대로 둔 채 같은 조건이라고 말하지 않습니다.

시간은 `19:00~19:50` 꼴이고 앞 두 자리만 비교하면 됩니다. 대상에는 `성인` · `청소년이상(14세 이상) …` · `8~13세` · `전연령`이 있습니다. 요일은 `월/수/금` · `화/목` · `월~금` 꼴입니다. 테스트 안에서 임시 데이터를 만들 때도 이 꼴을 씁니다.

예제/2주차_참고/채용/요구사항명세서.md 내려받기

# 예제 · 채용 (채용시험정보)

지난주 결과물이 없는 사람이 빌려 쓰는 예제입니다. 2주차 예제 B와 같은 재료(`채용.csv`)이고, 읽어 들이는 프로그램은 `app.py` 한 개입니다.

REQ-01 [상시] 시스템은 `채용.csv`의 공고 목록을 표시해야 한다.
REQ-02 [상시] 시스템은 각 줄에 모집기관 · 제목 · 상태 · 접수기간 · 등록일을 함께 표시해야 한다.

## 3주차에 더할 요구 후보

아직 구현되지 않은 것들입니다. 하나만 고르고, 그대로 쓰지 않고 내 말로 고쳐 `내소재.md` 3주차 칸 1)에 옮깁니다.

- [사건] 사용자가 상태를 입력할 때, 시스템은 그 상태의 공고만 표시해야 한다.
- [사건] 사용자가 정렬을 고를 때, 시스템은 등록일이 최근인 순서로 표시해야 한다.
- [원치 않는 동작] 만약 csv에 없는 상태를 입력하면, 시스템은 오류로 멈추지 않고 「해당 상태의 공고가 없습니다」를 표시하고 메뉴로 돌아가야 한다.
- [원치 않는 동작] 만약 접수기간이 `-`이면, 시스템은 날짜 계산을 시도하지 않고 「접수기간 없음」으로 표시해야 한다.

## 테스트에 쓸 값

상태는 `마감`과 `결과공고` 두 가지뿐입니다. `접수중`은 이 자료에 없으므로 경계 검사(없는 값을 넣었을 때)에 쓰기 좋습니다. 결과공고 행의 접수기간은 `-`입니다. 날짜가 들어가는 요구를 고르면 기준일을 인자로 받게 만들고 수업일 `2026-09-18`로 고정합니다. 그러지 않으면 내일 결과가 달라집니다.

# 예제 · 포트폴리오

SDD 포트폴리오가 없는 사람이 빌려 쓰는 예제입니다. SDD의 예제 D와 같은 재료(`index.html` 한 장)입니다. 인물 · 연락처는 가상입니다.

REQ-01 [상시] 시스템은 `index.html` 한 장에 소개 · 프로젝트 · 연락처를 표시해야 한다.

## 3주차에 더할 요구 후보

아직 구현되지 않은 것들입니다. 하나만 고르고, 그대로 쓰지 않고 내 말로 고쳐 `내소재.md` 3주차 칸 1)에 옮깁니다.

**다섯 중 둘은 판정할 수 없는 문장입니다.** 어느 것인지 골라내고, 고른 문장도 판정할 수 있는 말로 바꿔 씁니다.

1. [상시] 시스템은 연락처 유무와 관계없이 「연락처 공개 범위는 본인이 정합니다」 안내를 항상 표시해야 한다.
2. [상시] 시스템은 프로젝트 카드를 보기 좋게 정렬해야 한다.
3. [상시] 시스템은 프로젝트마다 기간과 역할을 함께 표시해야 한다.
4. [상시] 시스템은 첫 화면이 빠르게 뜨도록 해야 한다.
5. [상시] 시스템은 프로젝트마다 배포 주소 링크를 표시해야 한다.

## 테스트에 쓸 값

정상 검사는 `index.html`의 문자열(공개 이메일 · 프로젝트 이름 · 안내 문구)을 확인합니다. 경계 검사는 해당 부분을 지운 사본을 **테스트 안에서 임시 폴더에 만들어** 확인합니다. 파일을 따로 두지 않습니다. 버튼 동작은 브라우저에서 사람이 직접 검사하고 pytest 대상이 아닙니다.

`pytest`와 표준 라이브러리만 씁니다. `bs4` · `requests`는 서버에 없습니다.
