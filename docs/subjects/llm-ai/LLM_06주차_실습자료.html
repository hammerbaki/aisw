실행: .venv/bin/python chat.py

import os

import anthropic

MODEL = os.environ.get("COURSE_MODEL", "claude-haiku-4-5-20251001")
SYSTEM = "너는 친절한 도우미다. 두세 문장 이내로 간단히 답한다."

def ask(question: str) -> str:
    """질문을 보내고 답을 돌려준다."""

    # TODO 1: 이번 질문을 history에 넣는다.
    #         history.append({"role": "user", "content": question})

    # history가 비어 있으면 이번 질문만 보낸다 (TODO 1을 완성하기 전의 임시 동작)

    if STREAM:
        answer = ""
        print("AI: ", end="", flush=True)
        with client.messages.stream(
            model=MODEL,
            max_tokens=500,
            system=SYSTEM,
            messages=messages,
        ) as stream:
            for piece in stream.text_stream:
                print(piece, end="", flush=True)
                answer += piece
            usage = stream.get_final_message().usage
        print(f"\n   (input_tokens={usage.input_tokens}, output_tokens={usage.output_tokens})")
    else:
        response = client.messages.create(
            model=MODEL,
            max_tokens=500,
            system=SYSTEM,
            messages=messages,
        )
        answer = response.content[0].text
        print(f"AI: {answer}")
        print(f"   (input_tokens={response.usage.input_tokens}, output_tokens={response.usage.output_tokens})")

    # TODO 2: 이번 답을 history에 넣는다.
    #         history.append({"role": "assistant", "content": answer})

    return answer

        except EOFError:          # Ctrl+D
            question = ""
        if not question:
            print("종료합니다.")
            break
        ask(question)

# -*- coding: utf-8 -*-
"""임베딩 유사도 검색기 골격 (week06 실습 3 · 4용).

사용법:
    .venv/bin/python embed_search.py sim "문장1" "문장2"        # 두 문장의 유사도 출력
    .venv/bin/python embed_search.py index 학과FAQ.md           # 문서를 청킹·임베딩하여 index.json 생성
    .venv/bin/python embed_search.py search "질문 문장"          # 질문과 가까운 상위 3개 청크 출력
    .venv/bin/python embed_search.py eval 평가문항.csv           # 재현율(top-3) 측정

두 번째 문서(사내규정/복무규정.md)는 그 폴더에서 ../ 로 부릅니다.
    cd 사내규정 && ../.venv/bin/python ../embed_search.py index 복무규정.md

임베딩은 서버의 Ollama(공개 임베딩 모델)를 씁니다.
    - 환경변수 EMBED_MODEL: 임베딩 모델 이름 (과목 공지 값, 기본 nomic-embed-text)
    - 환경변수 CHUNK_MODE: "section"(제목 단위, 기본) 또는 "size"(고정 길이)
    - 환경변수 EMBED_BACKEND: "ollama"(기본) 또는 "fallback"
      임베딩 서버에 연결할 수 없으면 자동으로 예비 임베딩(fallback)으로 넘어갑니다.
      예비 임베딩은 설치가 필요 없는 저품질 대체 경로입니다. 검색은 되지만
      유사도 값과 재현율은 참고용입니다. 색인과 질문은 같은 경로로 만들어야 합니다.

index.json은 명령을 실행한 폴더에 만들어지고 읽힙니다.
cosine_similarity와 search는 실습에서 직접 완성합니다. 완성하면 tests/의 검색기 테스트 3개가 통과합니다.
"""
import csv
import json
import math
import os
import sys
import urllib.error
import urllib.request

OLLAMA_URL = os.environ.get("OLLAMA_URL", "http://localhost:11434")
EMBED_MODEL = os.environ.get("EMBED_MODEL", "nomic-embed-text")
CHUNK_MODE = os.environ.get("CHUNK_MODE", "section")
EMBED_BACKEND = os.environ.get("EMBED_BACKEND", "ollama")  # "ollama" 또는 "fallback"
INDEX_FILE = "index.json"
TOP_K = 3
_FALLBACK_DIM = 1024
_notified = {"fallback": False}


def _embed_fallback(text):
    """예비 임베딩: 글자 두 개씩 묶은 조각(bigram)을 해시해 고정 길이 벡터로 만든다.
    의미를 이해하지는 못하지만 표현이 겹치는 정도를 잰다. 서버가 없을 때만 쓴다."""
    text = " ".join(text.lower().split())
    vec = [0.0] * _FALLBACK_DIM
    grams = [text[i:i + 2] for i in range(len(text) - 1)] or [text]
    for g in grams:
        h = 0
        for ch in g:
            h = (h * 131 + ord(ch)) % 1000003
        vec[h % _FALLBACK_DIM] += 1.0
    norm = math.sqrt(sum(v * v for v in vec)) or 1.0
    return [v / norm for v in vec]


def _embed_ollama(text):
    """Ollama 임베딩 호출. 현재 주소(/api/embed)를 먼저 쓰고, 옛 판 서버면 /api/embeddings로 다시 부른다."""
    calls = (
        ("/api/embed", {"model": EMBED_MODEL, "input": text}, lambda data: data["embeddings"][0]),
        ("/api/embeddings", {"model": EMBED_MODEL, "prompt": text}, lambda data: data["embedding"]),
    )
    for path, body, pick in calls:
        request = urllib.request.Request(
            OLLAMA_URL + path,
            data=json.dumps(body).encode("utf-8"),
            headers={"Content-Type": "application/json"},
        )
        try:
            with urllib.request.urlopen(request, timeout=30) as response:
                return pick(json.loads(response.read()))
        except urllib.error.HTTPError as e:
            if e.code == 404 and path == "/api/embed":
                continue  # 옛 판 서버: 다음 주소로
            raise


def embed(text):
    """텍스트 하나를 임베딩 벡터(숫자 목록)로 바꾼다. (완성되어 있음)"""
    global EMBED_BACKEND
    if EMBED_BACKEND == "ollama":
        try:
            return _embed_ollama(text)
        except (urllib.error.URLError, OSError, KeyError, IndexError, ValueError) as e:
            EMBED_BACKEND = "fallback"
            print(f"[예비 경로] 임베딩 서버({OLLAMA_URL}, 모델 {EMBED_MODEL})에 연결할 수 없어 "
                  f"예비 임베딩으로 넘어갑니다. 유사도 값은 참고용입니다. ({type(e).__name__})",
                  file=sys.stderr)
    if not _notified["fallback"]:
        _notified["fallback"] = True
        if os.environ.get("EMBED_BACKEND", "ollama") == "fallback":
            print("[예비 경로] EMBED_BACKEND=fallback: 예비 임베딩을 씁니다. 유사도 값은 참고용입니다.",
                  file=sys.stderr)
    return _embed_fallback(text)


def chunk(text):
    """문서를 청크 목록으로 자른다. (완성되어 있음, 기준은 CHUNK_MODE로 선택)"""
    if CHUNK_MODE == "section":
        parts = []
        current = []
        for line in text.split("\n"):
            if line.startswith("## ") and current:
                parts.append("\n".join(current).strip())
                current = []
            current.append(line)
        if current:
            parts.append("\n".join(current).strip())
        return [p for p in parts if p]
    if CHUNK_MODE == "size":
        size = 150
        return [text[i:i + size] for i in range(0, len(text), size)]
    raise ValueError("CHUNK_MODE는 section 또는 size")

def build_index(path):
    with open(path, encoding="utf-8") as f:
        text = f.read()
    chunks = chunk(text)
    index = []
    for i, c in enumerate(chunks, 1):
        index.append({"text": c, "vector": embed(c)})
        print(f"임베딩 {i}/{len(chunks)}")
    with open(INDEX_FILE, "w", encoding="utf-8") as f:
        json.dump(index, f, ensure_ascii=False)
    print(f"청크 {len(chunks)}개를 {INDEX_FILE}에 저장 (CHUNK_MODE={CHUNK_MODE}, 임베딩={EMBED_BACKEND})")


def load_index():
    if not os.path.exists(INDEX_FILE):
        sys.exit(f"{INDEX_FILE}이 이 폴더에 없습니다. 먼저 index 명령으로 색인을 만듭니다.")
    with open(INDEX_FILE, encoding="utf-8") as f:
        return json.load(f)


def evaluate(csv_path):
    """평가 문항 CSV(질문,정답키워드)로 top-3 재현율을 측정한다."""
    with open(csv_path, encoding="utf-8") as f:
        rows = list(csv.DictReader(f))
    index = load_index()
    hit = 0
    for i, row in enumerate(rows, 1):
        results = search(row["질문"], index)
        found = any(row["정답키워드"] in text for _, text in results)
        hit += int(found)
        print(f"{i:2d}. [{'O' if found else 'X'}] {row['질문']}")
    print(f"\n재현율(top-{TOP_K}): {hit}/{len(rows)} ({hit / len(rows) * 100:.0f}%)")


def main():
    if len(sys.argv) < 2:
        print(__doc__)
        sys.exit(1)
    command = sys.argv[1]
    if command == "sim" and len(sys.argv) == 4:
        a, b = embed(sys.argv[2]), embed(sys.argv[3])
        print(f"유사도: {cosine_similarity(a, b):.4f}")
    elif command == "index" and len(sys.argv) == 3:
        build_index(sys.argv[2])
    elif command == "search" and len(sys.argv) == 3:
        for score, text in search(sys.argv[2], load_index()):
            print(f"\n[유사도 {score:.4f}] {text[:200]}")
    elif command == "eval" and len(sys.argv) == 3:
        evaluate(sys.argv[2])
    else:
        print(__doc__)
        sys.exit(1)


if __name__ == "__main__":
    main()

      임베딩 서버에 연결할 수 없으면 자동으로 예비 임베딩(fallback)으로 넘어갑니다.
      예비 임베딩은 설치가 필요 없는 저품질 대체 경로입니다. 검색은 되지만
      유사도 값과 재현율은 참고용입니다. 색인과 질문은 같은 경로로 만들어야 합니다.

index.json은 명령을 실행한 폴더에 만들어지고 읽힙니다.
cosine_similarity와 search는 실습에서 직접 완성합니다.
"""
import csv
import json
import math
import os
import sys
import urllib.error
import urllib.request

OLLAMA_URL = os.environ.get("OLLAMA_URL", "http://localhost:11434")
EMBED_MODEL = os.environ.get("EMBED_MODEL", "nomic-embed-text")
CHUNK_MODE = os.environ.get("CHUNK_MODE", "section")
EMBED_BACKEND = os.environ.get("EMBED_BACKEND", "ollama")  # "ollama" 또는 "fallback"
INDEX_FILE = "index.json"
TOP_K = 3
_FALLBACK_DIM = 1024
_notified = {"fallback": False}


def _embed_fallback(text):
    """예비 임베딩: 글자 두 개씩 묶은 조각(bigram)을 해시해 고정 길이 벡터로 만든다.
    의미를 이해하지는 못하지만 표현이 겹치는 정도를 잰다. 서버가 없을 때만 쓴다."""
    text = " ".join(text.lower().split())
    vec = [0.0] * _FALLBACK_DIM
    grams = [text[i:i + 2] for i in range(len(text) - 1)] or [text]
    for g in grams:
        h = 0
        for ch in g:
            h = (h * 131 + ord(ch)) % 1000003
        vec[h % _FALLBACK_DIM] += 1.0
    norm = math.sqrt(sum(v * v for v in vec)) or 1.0
    return [v / norm for v in vec]


def _embed_ollama(text):
    """Ollama 임베딩 호출. 현재 주소(/api/embed)를 먼저 쓰고, 옛 판 서버면 /api/embeddings로 다시 부른다."""
    calls = (
        ("/api/embed", {"model": EMBED_MODEL, "input": text}, lambda data: data["embeddings"][0]),
        ("/api/embeddings", {"model": EMBED_MODEL, "prompt": text}, lambda data: data["embedding"]),
    )
    for path, body, pick in calls:
        request = urllib.request.Request(
            OLLAMA_URL + path,
            data=json.dumps(body).encode("utf-8"),
            headers={"Content-Type": "application/json"},
        )
        try:
            with urllib.request.urlopen(request, timeout=30) as response:
                return pick(json.loads(response.read()))
        except urllib.error.HTTPError as e:
            if e.code == 404 and path == "/api/embed":
                continue  # 옛 판 서버: 다음 주소로
            raise


def embed(text):
    """텍스트 하나를 임베딩 벡터(숫자 목록)로 바꾼다. (완성되어 있음)"""
    global EMBED_BACKEND
    if EMBED_BACKEND == "ollama":
        try:
            return _embed_ollama(text)
        except (urllib.error.URLError, OSError, KeyError, IndexError, ValueError) as e:
            EMBED_BACKEND = "fallback"
            print(f"[예비 경로] 임베딩 서버({OLLAMA_URL}, 모델 {EMBED_MODEL})에 연결할 수 없어 "
                  f"예비 임베딩으로 넘어갑니다. 유사도 값은 참고용입니다. ({type(e).__name__})",
                  file=sys.stderr)
    if not _notified["fallback"]:
        _notified["fallback"] = True
        if os.environ.get("EMBED_BACKEND", "ollama") == "fallback":
            print("[예비 경로] EMBED_BACKEND=fallback: 예비 임베딩을 씁니다. 유사도 값은 참고용입니다.",
                  file=sys.stderr)
    return _embed_fallback(text)


def chunk(text):
    """문서를 청크 목록으로 자른다. (완성되어 있음, 기준은 CHUNK_MODE로 선택)"""
    if CHUNK_MODE == "section":
        parts = []
        current = []
        for line in text.split("\n"):
            if line.startswith("## ") and current:
                parts.append("\n".join(current).strip())
                current = []
            current.append(line)
        if current:
            parts.append("\n".join(current).strip())
        return [p for p in parts if p]
    if CHUNK_MODE == "size":
        size = 150
        return [text[i:i + size] for i in range(0, len(text), size)]
    raise ValueError("CHUNK_MODE는 section 또는 size")


def cosine_similarity(a, b):
    """두 벡터의 코사인 유사도를 반환한다. [실습에서 완성]

    공식: (a와 b의 내적) / (a의 크기 * b의 크기)
    내적은 같은 위치 성분끼리 곱해 모두 더한 값,
    크기는 각 성분 제곱합의 제곱근(math.sqrt)이다.
    """
    dot = sum(x * y for x, y in zip(a, b))  # zip: 두 목록을 같은 위치끼리 짝지음
    norm_a = math.sqrt(sum(x * x for x in a))
    norm_b = math.sqrt(sum(x * x for x in b))
    return dot / (norm_a * norm_b)


def search(query, index):
    """질문과 가까운 순으로 (유사도, 청크) 상위 TOP_K개를 반환한다. [실습에서 완성]

    순서: ① 질문을 embed로 벡터화 ② index의 모든 청크 벡터와
    cosine_similarity 계산 ③ 유사도 내림차순 정렬 후 앞 TOP_K개 반환.
    index는 [{"text": 청크, "vector": 벡터}, ...] 형태다.
    """
    query_vector = embed(query)
    scored = [(cosine_similarity(query_vector, item["vector"]), item["text"])
              for item in index]
    scored.sort(key=lambda pair: -pair[0])  # lambda는 이름 없이 한 줄로 쓰는 임시 함수. 점수가 큰 순서가 되도록 음수로 뒤집어 정렬
    return scored[:TOP_K]


def build_index(path):
    with open(path, encoding="utf-8") as f:
        text = f.read()
    chunks = chunk(text)
    index = []
    for i, c in enumerate(chunks, 1):
        index.append({"text": c, "vector": embed(c)})
        print(f"임베딩 {i}/{len(chunks)}")
    with open(INDEX_FILE, "w", encoding="utf-8") as f:
        json.dump(index, f, ensure_ascii=False)
    print(f"청크 {len(chunks)}개를 {INDEX_FILE}에 저장 (CHUNK_MODE={CHUNK_MODE}, 임베딩={EMBED_BACKEND})")


def load_index():
    if not os.path.exists(INDEX_FILE):
        sys.exit(f"{INDEX_FILE}이 이 폴더에 없습니다. 먼저 index 명령으로 색인을 만듭니다.")
    with open(INDEX_FILE, encoding="utf-8") as f:
        return json.load(f)


def evaluate(csv_path):
    """평가 문항 CSV(질문,정답키워드)로 top-3 재현율을 측정한다."""
    with open(csv_path, encoding="utf-8") as f:
        rows = list(csv.DictReader(f))
    index = load_index()
    hit = 0
    for i, row in enumerate(rows, 1):
        results = search(row["질문"], index)
        found = any(row["정답키워드"] in text for _, text in results)
        hit += int(found)
        print(f"{i:2d}. [{'O' if found else 'X'}] {row['질문']}")
    print(f"\n재현율(top-{TOP_K}): {hit}/{len(rows)} ({hit / len(rows) * 100:.0f}%)")


def main():
    if len(sys.argv) < 2:
        print(__doc__)
        sys.exit(1)
    command = sys.argv[1]
    if command == "sim" and len(sys.argv) == 4:
        a, b = embed(sys.argv[2]), embed(sys.argv[3])
        print(f"유사도: {cosine_similarity(a, b):.4f}")
    elif command == "index" and len(sys.argv) == 3:
        build_index(sys.argv[2])
    elif command == "search" and len(sys.argv) == 3:
        for score, text in search(sys.argv[2], load_index()):
            print(f"\n[유사도 {score:.4f}] {text[:200]}")
    elif command == "eval" and len(sys.argv) == 3:
        evaluate(sys.argv[2])
    else:
        print(__doc__)
        sys.exit(1)


if __name__ == "__main__":
    main()

# AI소프트웨어학과 실습 FAQ (실습용 가상 문서)

RAG 실습의 공통 검색 대상 문서입니다. 실제 학과 공지가 아니라 실습용으로 작성된 가상 FAQ입니다.

## 실습 서버에는 어떻게 접속하나요

안내된 방법으로 과목 실습 서버에 터미널로 접속해 발급받은 계정으로 로그인하면 개인 작업 공간이 열립니다. 실습은 서버에서 실행되므로 개인 컴퓨터 사양과 무관하며, 실습실과 집 어디에서든 같은 환경을 사용할 수 있습니다. 접속이 안 되면 먼저 학과 공지 채널의 점검 일정을 확인합니다.

## API 키를 잃어버렸어요

API 키는 재발급할 수 있습니다. 교수에게 계정 이름과 함께 재발급을 요청하면 기존 키는 회수되고 새 키가 발급됩니다. 새 키는 홈 폴더의 환경 설정 파일에 다시 등록해야 합니다.

## 산출물은 어디에 제출하나요

제출물은 별도 업로드 없이 개인 폴더의 상태로 제출을 갈음합니다. 수업 종료 시각 기준으로 개인 폴더의 해당 week 폴더에 산출물이 있으면 제출로 인정됩니다. 파일 이름은 강의안의 제출물 목록을 따릅니다.

## 결석하면 실습 점수는 어떻게 되나요

제출물은 수업 참여를 전제로 하므로 결석 시 해당 회차 참여 점수가 빠집니다. 병결 등 증빙이 있는 결석은 다음 수업 전까지 보충 제출을 인정합니다. 보충 제출 방법은 교수에게 개별 문의합니다.

## 지속 프로젝트를 바꾸고 싶어요

지속 프로젝트는 학기 중 한 번에 한해 변경할 수 있습니다. 변경 사유와 새 프로젝트 한 줄 소개를 개인 폴더의 지속프로젝트.md에 기록하고 교수에게 알리면 됩니다. 이미 제출한 기록은 그대로 평가에 반영됩니다.

## 시험 범위는 어디까지인가요

개념 시험의 범위는 시험 직전 회차까지의 강의안 전체입니다. 강의안의 확인 문제와 각 회차의 학습 목표가 출제 기준이 됩니다. 실습 코드의 세부 문법보다 원리와 판단 근거를 묻습니다.

## 개인 폴더는 백업되나요

개인 폴더는 서버에서 주기적으로 백업되지만, 중요한 산출물은 각자 git 저장소에 커밋해 두는 것을 권장합니다. 삭제 복구 요청은 교수에게 폴더 경로와 삭제 추정 시각을 알려 접수합니다.

## Ollama에 다른 모델을 추가하고 싶어요

서버의 공개 모델은 학과가 사전 배포한 공유 캐시에서 제공됩니다. 추가하고 싶은 모델이 있으면 모델 이름과 용도를 적어 교수에게 요청하면 검토 후 캐시에 추가됩니다. 개별 다운로드는 디스크 용량 문제로 제한됩니다.

## 성적에 이의가 있으면 어떻게 하나요

성적 공시 후 일주일 안에 이의신청서를 작성해 교수에게 제출합니다. 이의신청서에는 해당 항목, 본인이 확인한 제출물 경로, 이의 사유를 적습니다. 확인 후 결과를 개별 통보합니다.

## 이 과목은 졸업작품과 연계되나요

지속 프로젝트를 졸업작품의 사전 작업으로 삼을 수 있습니다. 기말 개인 프로젝트의 명세서와 산출물은 졸업작품 계획서의 재료로 재사용이 가능하며, 희망자는 학기 말 상담을 신청합니다.

## 개인 노트북 사양이 낮아도 되나요

실습은 모두 서버에서 실행되므로 브라우저가 동작하는 노트북이면 충분합니다. 로컬 모델 실행도 서버 컨테이너 안에서 이루어지므로 개인 노트북의 메모리나 GPU는 요구되지 않습니다.

## 수강 정정 기간 이후에 합류했어요

합류 즉시 교수에게 계정과 API 키 발급을 요청하고, 오리엔테이션 회차 강의안의 환경 점검 실습을 스스로 수행합니다. 지난 회차의 제출물은 다음 수업 전까지 개인 폴더에 채워 보충 제출합니다.

# 복무 규정 (발췌 · 실습용 가상 문서)

RAG 실습의 두 번째 검색 대상 문서입니다. 실제 회사의 규정이 아니라 실습용으로 작성한 가상 회사의 복무 규정에서 조항 일부를 뽑은 것입니다. 이 문서는 2026년 3월 1일 개정판이며, 조항 번호는 원문 번호를 그대로 두었습니다.

## 제3조 (근무시간)

근무시간은 오전 9시부터 오후 6시까지이며, 점심시간은 12:00부터 13:00까지입니다. 1주 근무시간은 40시간을 기준으로 합니다. 업무상 필요로 근무시간을 바꿀 때는 부서장이 미리 공지합니다.

## 제7조 (유연근무)

직원은 오전 8시부터 10시 사이에서 출근 시각을 골라 하루 8시간을 근무할 수 있습니다. 오전 10시부터 오후 4시까지의 코어타임에는 모든 직원이 근무해야 합니다. 유연근무는 적용 시작일 한 달 전까지 인사 시스템에서 신청합니다.

## 제12조 (지각과 조퇴)

출근 시각을 넘겨 도착하면 지각으로 처리합니다. 지각한 날에는 당일 안에 사유를 인사 시스템에 입력합니다. 조퇴는 부서장의 사전 승인을 받아야 하며, 긴급한 경우에는 사후 승인을 받습니다.

## 제15조 (연차휴가)

1년간 80% 이상 출근한 직원에게 15일의 연차휴가를 줍니다. 3년 이상 근속한 직원에게는 2년마다 1일을 더하며, 연차휴가는 최대 25일입니다. 연차휴가는 사용 3일 전까지 인사 시스템에서 신청하고, 반나절 단위(반차)로도 쓸 수 있습니다.

## 제16조 (경조사 휴가)

본인 결혼은 5일, 자녀 결혼은 1일, 부모와 배우자의 부모 사망은 5일, 조부모 사망은 3일의 경조사 휴가를 줍니다. 경조사 휴가는 연차휴가에서 차감하지 않습니다. 증빙 서류는 휴가가 끝난 뒤 7일 안에 제출합니다.

## 제18조 (병가)

질병이나 부상으로 근무할 수 없는 직원은 연간 10일까지 유급 병가를 쓸 수 있습니다. 병가가 연속 3일을 넘으면 의사의 진단서를 제출해야 합니다. 병가는 당일 오전 10시 전까지 부서장에게 알립니다.

## 제21조 (출장)

출장을 가려면 출발 전에 출장 신청서를 작성해 부서장의 승인을 받습니다. 국내 출장의 숙박비는 1박 10만 원을 상한으로 실비를 지급합니다. 출장이 끝나면 5일 안에 결과 보고서를 제출합니다.

## 제24조 (재택근무)

재택근무는 부서장 승인 후 인사 시스템에서 신청합니다. 재택근무는 주 2일 이내로 할 수 있습니다. 재택근무 중에는 회사가 지급한 장비로 VPN에 접속해 업무를 봅니다.

## 제27조 (업무용 장비와 보안)

회사가 지급한 노트북과 휴대 장비는 업무 용도로만 씁니다. 장비를 잃어버리거나 도난당하면 24시간 이내에 보안팀에 신고합니다. 개인 USB 저장장치에는 회사 자료를 저장하지 않습니다.

## 제30조 (교육 지원)

직무와 관련된 외부 교육은 1인당 연 100만 원 한도에서 교육비를 지원합니다. 교육을 신청하려면 교육 시작 2주 전까지 교육 계획서를 제출합니다. 교육이 끝나면 수료증을 제출해야 교육비를 받을 수 있습니다.

## 제33조 (경비 정산)

업무에 쓴 비용은 법인카드로 결제하는 것이 원칙입니다. 개인 카드로 결제했다면 영수증을 첨부해 사용일로부터 30일 이내에 정산을 신청합니다. 영수증이 없는 비용은 정산하지 않습니다.

## 제36조 (육아휴직)

만 8세 이하 자녀를 둔 직원은 자녀 한 명당 최대 1년의 육아휴직을 쓸 수 있습니다. 육아휴직은 시작 예정일 30일 전까지 인사팀에 신청서를 제출합니다. 육아휴직 기간은 근속 기간에 포함합니다.
