# -*- coding: utf-8 -*-
"""평가 러너 (week11 실습용, 완성되어 있음. 실행하며 출력을 읽는 용도).

같은 챗봇 응답 묶음을 두 방식으로 채점하여 비교합니다.
    ① 규칙 채점: 정답셋의 키워드가 응답에 들어 있는가 (빠르고 싸지만 융통성 없음)
    ② Judge 채점: Claude가 기준을 들고 응답을 판정 (융통성 있지만 비용·비결정성)

사용법:
    python eval_runner.py --rule-only     # 규칙 채점만 (API 불필요)
    python eval_runner.py                 # 규칙 + Judge 채점 비교

전제(Judge 사용 시): 환경변수 ANTHROPIC_API_KEY, COURSE_MODEL
"""
import json
import os
import sys

HERE = os.path.dirname(os.path.abspath(__file__))

JUDGE_SYSTEM = """너는 학과 FAQ 챗봇의 응답을 채점하는 심판이다. 기준은 세 가지다.
① 정확성: 참고 답안의 사실과 어긋나지 않는가.
② 거절 적절성: 문서에 없는 주제라면 지어내지 않고 거절했는가.
③ 근거 없는 단정 금지: 참고 답안에 없는 절차나 조건을 만들어 붙이지 않았는가.
반드시 첫 줄에 "판정: 통과" 또는 "판정: 탈락"을 쓰고, 둘째 줄에 "이유: "로 한 문장을 쓴다."""


def rule_grade(answer, keywords):
    """키워드가 하나라도 없으면 탈락하는 단순 규칙."""
    return all(k in answer for k in keywords)


def judge_grade(client, model, item, answer):
    prompt = (f"질문: {item['question']}\n참고 답안: {item['reference']}\n"
              f"챗봇 응답: {answer}\n채점하라.")
    r = client.messages.create(model=model, max_tokens=200,
                               system=JUDGE_SYSTEM,
                               messages=[{"role": "user", "content": prompt}])
    text = "".join(b.text for b in r.content if b.type == "text")
    verdict = "통과" in text.split("\n")[0]
    reason = next((ln[3:].strip() for ln in text.split("\n") if ln.startswith("이유:")), text[:80])
    return verdict, reason


def main():
    rule_only = "--rule-only" in sys.argv
    items = json.load(open(os.path.join(HERE, "정답셋.json"), encoding="utf-8"))
    answers = json.load(open(os.path.join(HERE, "응답로그.json"), encoding="utf-8"))

    client = model = None
    if not rule_only:
        import anthropic
        client = anthropic.Anthropic()
        model = os.environ.get("COURSE_MODEL", "claude-haiku-4-5")

    print(f"{'번호':<4}{'규칙':<6}{'Judge':<7}비고")
    mismatch = []
    for item in items:
        answer = answers[str(item["id"])]
        rule = rule_grade(answer, item["keywords"])
        if rule_only:
            print(f"{item['id']:<4}{'통과' if rule else '탈락':<6}{'-':<7}")
            continue
        judge, reason = judge_grade(client, model, item, answer)
        note = ""
        if rule != judge:
            note = f"불일치! {reason}"
            mismatch.append(item["id"])
        print(f"{item['id']:<4}{'통과' if rule else '탈락':<6}{'통과' if judge else '탈락':<7}{note}")

    if not rule_only:
        print(f"\n불일치 문항: {mismatch if mismatch else '없음'}")
        print("불일치 문항의 응답과 정답셋을 직접 읽고, 어느 채점이 옳았는지 판단해 기록하세요.")


if __name__ == "__main__":
    main()

# -*- coding: utf-8 -*-
"""재시도 상한(MAX_RETRIES) 안전장치 (week11 실습용, 완성되어 있음).

자동 반복 작업의 기본 안전장치입니다. 생성과 검증을 반복하되,
상한을 넘으면 무한 반복 대신 실패를 보고하고 사람에게 넘깁니다.

사용법:
    python retry_loop.py success   # 2번째 시도에서 검증 통과
    python retry_loop.py fail      # 상한 도달 후 수동 개입 요청

API 키가 필요 없습니다 (생성기와 검증기는 시나리오용 대역).
"""
import sys

MAX_RETRIES = 3


def generate(scenario, attempt):
    """생성 단계의 대역. 실제로는 AI 생성 호출이 들어갈 자리다."""
    if scenario == "success" and attempt >= 2:
        return f"산출물_v{attempt}_정상"
    return f"산출물_v{attempt}_불량"


def verify(artifact):
    """검증 단계의 대역. 실제로는 pytest나 배포 전 점검이 들어갈 자리다."""
    ok = "정상" in artifact
    reason = "" if ok else "검증 실패: 기준 미달"
    return ok, reason


def generate_with_retries(scenario):
    for attempt in range(1, MAX_RETRIES + 1):
        artifact = generate(scenario, attempt)
        ok, reason = verify(artifact)
        print(f"[시도 {attempt}/{MAX_RETRIES}] 산출물={artifact} 통과={ok} {reason}")
        if ok:
            return artifact
    return None


def main():
    scenario = sys.argv[1] if len(sys.argv) > 1 else "success"
    if scenario not in ("success", "fail"):
        print("사용법: python retry_loop.py success|fail")
        return 1
    result = generate_with_retries(scenario)
    if result is None:
        print(f"중단: {MAX_RETRIES}회 실패. 수동 개입이 필요합니다.")
        return 1
    print(f"완료: {result}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

# 11주차 검증 키트

검증과 평가(Evals), 검토자 서브에이전트 실습용 키트입니다. 개인 폴더의 `week11/` 아래에 복사하여 사용합니다. `reviewer/`와 `deploy-check/`의 구도는 참고 서적 《하네스 엔지니어링 with 클로드 코드》 공식 예제 저장소(https://github.com/revfactory/harness-engineering-with-cc, Apache 2.0)의 ex-04-01(읽기 전용 검토자), ex-11-06(의도 버그 검증)을 수업용으로 각색한 것입니다.

## 구성

| 폴더·파일 | 내용 |
|---|---|
| `evals/eval_runner.py` | 규칙 채점과 Judge 채점을 비교하는 평가 러너 (완성되어 있음) |
| `evals/정답셋.json`, `evals/응답로그.json` | 채점 재료: 질문·키워드·참고 답안 6건과 챗봇 응답 로그 |
| `reviewer/검토자_골격.md` | 배포 전 점검 검토자(서브에이전트) 정의 파일 골격 |
| `deploy-check/site/` | 배포 직전 폴더 샘플. 위반 3건이 심어져 있음 (검토자 성능 검증용) |
| `deploy-check/정답_위반목록.md` | 심어 둔 위반의 정답. **판정 기록을 마친 뒤에 엽니다** |

## 사용 순서

```bash
python evals/eval_runner.py --rule-only   # 규칙 채점만 (API 불필요)
python evals/eval_runner.py               # 규칙 + Judge 비교
```

검토자 정의 파일은 골격을 참고해 `week11/.claude/agents/deploy-checker.md`에 직접 작성합니다. 절차는 강의안을 따릅니다.

## 주의

- `deploy-check/site/`의 키 문자열은 실습용 가짜 값이며, 이 폴더는 절대 실제로 배포하지 않습니다.
- `정답_위반목록.md`는 자기 판정 기록을 마치기 전에 열지 않습니다.

## 전제

`eval_runner.py`의 Judge 채점만 API 환경(`ANTHROPIC_API_KEY`, `COURSE_MODEL`)이 필요합니다. 나머지는 API 키가 필요 없습니다.

# 11주차 자동화 키트

업무 자동화와 배포 파이프라인 실습용 키트입니다. 개인 폴더의 `week11/` 아래에 복사하여 사용합니다. `retry/retry_loop.py`는 참고 서적 《하네스 엔지니어링 with 클로드 코드》 공식 예제 저장소(https://github.com/revfactory/harness-engineering-with-cc, Apache 2.0)의 ex-08-15를 수업용으로 각색한 것입니다.

## 구성

| 폴더·파일 | 내용 |
|---|---|
| `pipeline/서버로그.txt`, `pipeline/pipeline.sh` | 수집·요약·통지 파이프라인 골격. 종료 코드 분기를 실습에서 완성 |
| `hooks/훅_설정예시.md` | PostToolUse 훅 설정 예시와 주의점 |
| `deploy/deploy.sh` | 배포 파이프라인: 점검 2종 통과 시에만 wrangler 배포. 배포 명령 연결을 실습에서 완성 |
| `retry/retry_loop.py` | 재시도 상한(MAX_RETRIES) 안전장치. 성공·실패 2 시나리오 (완성되어 있음, API 불필요) |
| `sync/sync_체크리스트.md` | 코드-문서 동기화 게이트 점검표 |
| `크로스테스트_기록표.md` | 동료 산출물 시험 기록표 |

## 전제

1. `pipeline.sh`는 Claude Code 로그인 환경(학과 키, Haiku)이 필요합니다.
2. `deploy.sh`의 배포 단계는 Cloudflare 계정과 wrangler 실행 환경(Node.js)이 필요합니다. 환경이 없으면 점검 단계까지 실행하고 배포는 대시보드 업로드(배포 카드)로 대체합니다.
3. `retry_loop.py`는 API 키가 필요 없습니다.

# PostToolUse 훅 설정 예시

프로젝트의 `.claude/settings.json`에 아래 hooks 절을 추가하면, Claude Code가 파일을 쓰거나 고칠 때마다 파이썬 문법 검사가 자동 실행됩니다. 검사가 실패하면 종료 코드 2와 오류 출력(stderr)이 Claude에게 전달되어 스스로 대처합니다. 통과 시에는 아무것도 출력하지 않습니다. 통과 출력(종료 코드 0의 stdout)은 Claude에게 전달되지 않는 것이 훅의 규약입니다.

```json
{
  "hooks": {
    "PostToolUse": [
      {
        "matcher": "Write|Edit",
        "hooks": [
          {
            "type": "command",
            "command": "python -m compileall -q . || { echo '훅: 문법 오류 발견. 수정할 것' >&2; exit 2; }"
          }
        ]
      }
    ]
  }
}
```

- 훅은 지시가 아니라 이벤트마다 반드시 실행되는 강제 장치입니다.
- 검사 명령은 프로젝트에 맞게 바꿉니다 (예: `pytest -q`).
- 훅 설정 형식은 버전에 따라 바뀔 수 있습니다. `/hooks`와 [공식 문서](https://code.claude.com/docs/en/hooks-guide)가 우선합니다.
- 설정 변경 후에는 세션을 재시작해야 반영됩니다.

## 통지형 훅 (심화 선택)

검사형 훅이 결과를 대화 컨텍스트로 돌려준다면, 통지형 훅은 결과를 밖(메신저, 모바일)으로 내보냅니다. 예를 들어 긴 작업이 끝날 때 Slack 웹훅으로 알림을 보낼 수 있습니다.

```json
{
  "type": "command",
  "command": "curl -s -X POST \"$SLACK_WEBHOOK_URL\" -H 'Content-type: application/json' --data '{\"text\":\"Claude Code: 작업 완료\"}'"
}
```

- 웹훅 URL도 API 키와 같은 비밀값입니다. 설정 파일에 직접 쓰지 않고 환경변수(`SLACK_WEBHOOK_URL`)로 둡니다.
- 개인 Slack 워크스페이스와 웹훅 발급이 필요하므로 심화 선택입니다. 수업의 기본 실습은 파일 기록 방식의 통지를 유지합니다.
- 완성 예시는 [클로드 코드 마스터 공식 저장소의 slack-notification 샘플](https://github.com/claude-code-expert/lecture/tree/main/samples/slack-notification)에 있습니다(CC BY-NC-ND 4.0, 링크 열람용. 위 예시는 자체 작성입니다).
