# -*- coding: utf-8 -*-
"""주문 관리 샘플 모듈 (CLAUDE.md 실습용)."""

_ORDERS = {}
_STOCK = {"노트북": 5, "마우스": 20, "키보드": 0}


def create_order(order_id, product, quantity):
    """주문을 생성한다. 재고가 부족하면 ValueError를 던진다."""
    if quantity <= 0:
        raise ValueError("수량은 1 이상이어야 한다")
    stock = _STOCK.get(product, 0)
    if quantity > stock:
        raise ValueError("재고 부족")
    _STOCK[product] = stock - quantity
    _ORDERS[order_id] = {"product": product, "quantity": quantity, "status": "접수"}
    return _ORDERS[order_id]


def get_order(order_id):
    """주문을 조회한다."""
    if order_id not in _ORDERS:
        raise KeyError("존재하지 않는 주문")
    return _ORDERS[order_id]


def reset():
    """테스트용 초기화."""
    _ORDERS.clear()
    _STOCK.clear()
    _STOCK.update({"노트북": 5, "마우스": 20, "키보드": 0})

"""5주차 실습 3의 1단계: 문장의 토큰 수를 잰다.

사용법:
    python3 스크립트/token_count.py "측정할 문장"
    python3 스크립트/token_count.py --check        # 키·모델 설정만 확인

동작:
    4주차 채점기(ab_runner.py)와 같은 방식으로 파이썬 표준 라이브러리(urllib)만 써서
    Claude API의 토큰 계산 기능(count_tokens)에 문장을 보내고 토큰 수를 받는다.
    설치할 패키지가 없다. 인증은 환경변수 ANTHROPIC_API_KEY, 모델은 환경변수
    COURSE_MODEL(없으면 수업 기본 모델)을 쓴다. 키는 화면에 출력하지 않는다.

참고:
    결과에는 요청 형식에 붙는 고정 토큰이 더해져 있다(교수 화면 7토큰. "a" 한 글자가 8).
    측정값 = 고정분 + 문장 토큰. 비교할 때는 고정분을 빼고 본다.
"""
import json
import os
import sys
import urllib.error
import urllib.request

MODEL = os.environ.get("COURSE_MODEL", "claude-haiku-4-5-20251001")
URL = "https://api.anthropic.com/v1/messages/count_tokens"


def count_tokens(text):
    key = os.environ.get("ANTHROPIC_API_KEY", "")
    if not key:
        raise SystemExit("KEY_NOT_SET: ANTHROPIC_API_KEY가 없습니다. 강의안 0장의 키 등록을 먼저 합니다.")
    payload = json.dumps({"model": MODEL, "messages": [{"role": "user", "content": text}]}).encode()
    headers = {"x-api-key": key, "anthropic-version": "2023-06-01", "content-type": "application/json"}
    for attempt in range(2):
        req = urllib.request.Request(URL, data=payload, headers=headers)
        try:
            with urllib.request.urlopen(req, timeout=30) as resp:
                return json.load(resp)["input_tokens"]
        except urllib.error.HTTPError as e:
            if e.code == 400 and attempt == 0 and "anthropic-beta" not in headers:
                headers["anthropic-beta"] = "token-counting-2024-11-01"  # 구 버전 API용. 보통은 필요 없음
                continue
            raise SystemExit(f"HTTP {e.code}: 인증·모델 권한·요청 제한을 확인합니다. 원문 오류에 키가 들어 있을 수 있어 출력하지 않습니다.")
        except (urllib.error.URLError, TimeoutError):
            raise SystemExit("네트워크 오류 또는 시간 초과. 잠시 뒤 다시 실행합니다.")


if __name__ == "__main__":
    args = sys.argv[1:]
    if args == ["--check"]:
        print("MODEL", MODEL, "API_KEY_SET=" + str(bool(os.environ.get("ANTHROPIC_API_KEY"))))
        sys.exit(0)
    if not args:
        print('사용법: python3 스크립트/token_count.py "측정할 문장"')
        sys.exit(1)
    text = " ".join(args)
    print(f"문장: {text}")
    print(f"토큰 수: {count_tokens(text)}")

# -*- coding: utf-8 -*-
"""포트폴리오 고정 테스트. 고치지 않는다(TDD에서 동결한 테스트와 같은 역할).

REQ-01 섹션 id 네 개 · REQ-03 이메일은 프로필.md에 있는 것만 · REQ-04 전화번호 없음.
pytest와 표준 라이브러리만 쓴다.
"""
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
PAGE = (ROOT / "index.html").read_text(encoding="utf-8")
PROFILE = (ROOT / "프로필.md").read_text(encoding="utf-8")


def test_섹션_id_네_개가_있다():
    for sid in ("about", "skills", "projects", "contact"):
        assert f'id="{sid}"' in PAGE, f"섹션 id {sid}가 없습니다"


def test_이메일은_프로필에_있는_것만_쓴다():
    emails = set(re.findall(r"[\w.+-]+@[\w-]+(?:\.[\w-]+)+", PAGE))
    assert emails, "이메일이 없습니다"
    for e in emails:
        assert e in PROFILE, f"프로필.md에 없는 이메일입니다: {e}"


def test_전화번호를_표시하지_않는다():
    found = re.search(r"01[016789][-. ]?\d{3,4}[-. ]?\d{4}", PAGE)
    assert found is None, f"전화번호가 있습니다: {found.group() if found else ''}"

## 왜 「200줄 이내」와 「없는 것은 넣지 마라」가 있는가

CLAUDE.md는 세션마다 컨텍스트에 실려 토큰 비용을 반복해서 내고, 너무 길면 놓칩니다. 골격의 규칙(app/ 아래에 둔다)을 그대로 두면 포트폴리오에는 app/이 없어 규칙이 헛돕니다. 「모든 세션에 적용되는가」를 통과한 것만 남깁니다.

# 주문 관리 샘플 프로젝트 명세서

CLAUDE.md 실습의 재료입니다. 리버스 엔지니어링 회차에서 배운 EARS(Easy Approach to Requirements Syntax, 정형 요구사항 문형 기법) 문형을 따르며, CLAUDE.md 작성 실습의 원천 자료로 사용합니다.

## 요구사항

- 사용자가 주문을 생성하면, 시스템은 재고를 차감하고 주문 상태를 "접수"로 기록해야 한다.
- 재고보다 많은 수량이 요청되면, 시스템은 주문을 거절하고 오류를 반환해야 한다.
- 수량이 1 미만으로 요청되면, 시스템은 주문을 거절해야 한다.
- 존재하지 않는 주문이 조회되면, 시스템은 오류를 반환해야 한다.

## 구조 규칙

- `app/`은 업무 규칙만 담고 외부 라이브러리에 의존하지 않는다.
- 모든 기능 변경은 `tests/`의 테스트와 함께 제출한다.
- 함수 이름은 동사로 시작한다.

## 실행

- 테스트: `~/week05/.venv/bin/python -m pytest -q`

# 요구사항명세서 · 포트폴리오

CLAUDE.md 실습(실습 1 · 8/9~9/9)의 두 번째 프로젝트입니다. `index.html` 한 장짜리 정적 페이지이고, 인물 · 연락처는 가상입니다(`프로필.md`). 골격(주문 관리 · Python)과 종류가 다른 프로젝트에 규칙 파일을 옮겨 보는 것이 목적입니다.

## 요구사항

REQ-01 [상시] 시스템은 `index.html` 한 장에 소개 · 기술 · 프로젝트 · 연락처를 표시해야 한다. 네 섹션의 id는 about · skills · projects · contact이다.
REQ-02 [사건] 사용자가 프로젝트 카드를 누를 때, 시스템은 상세 설명을 펼쳐야 한다.
REQ-03 [상시] 시스템은 `프로필.md`에 있는 연락처만 표시해야 한다.
REQ-04 [상시] 시스템은 전화번호와 집 주소를 표시하지 않아야 한다.

## 테스트

`tests/test_page.py`의 테스트 세 개가 REQ-01 · REQ-03 · REQ-04를 확인합니다. **고정 테스트이므로 고치지 않습니다.** TDD에서 동결한 테스트와 같은 역할입니다. REQ-02(카드 펼치기)는 브라우저에서 사람이 확인합니다.

`pytest`와 표준 라이브러리만 씁니다.
