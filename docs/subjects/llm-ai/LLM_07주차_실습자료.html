# -*- coding: utf-8 -*-
"""도구 호출 사이클 관찰 스크립트 (week07 교수 시연 1용, 완성되어 있음).

사용법:
    .venv/bin/python calc_tool.py "공급가액 100만 원의 부가세는?"
    .venv/bin/python calc_tool.py "부가세는 언제 생긴 제도야?"

도구가 필요한 질문과 필요 없는 질문을 각각 던져,
stop_reason과 사이클(도구 정의 전달 → 호출 결정 → 실행 → 결과 재주입 → 최종 답)을 관찰합니다.

전제: 환경변수 ANTHROPIC_API_KEY. COURSE_MODEL은 없으면 기본값을 씁니다.
"""
import os
import sys

import anthropic

MODEL = os.environ.get("COURSE_MODEL", "claude-haiku-4-5-20251001")
MAX_ROUNDS = 5  # 도구 호출 반복의 상한 (무한 반복을 막는다)

_client = None


def get_client():
    """API 클라이언트를 처음 쓸 때 만든다(키는 환경변수 ANTHROPIC_API_KEY에서 읽는다)."""
    global _client
    if _client is None:
        _client = anthropic.Anthropic()
    return _client


# 도구 정의: 이름(영문), 설명, 입력 스키마. 모델은 이 정의만 보고 호출 여부를 판단한다.
VAT_TOOL = {
    "name": "vat_calc",
    "description": "공급가액(원)을 받아 부가세 10%와 합계 금액을 계산한다. "
                   "금액 계산이 필요한 질문에 사용한다. "
                   "부가세 제도나 세법을 설명하는 질문에는 사용하지 않는다.",
    "input_schema": {
        "type": "object",
        "properties": {
            "amount": {"type": "number", "description": "공급가액(원, 부가세 제외 금액)"}
        },
        "required": ["amount"],
    },
}


def run_tool(name, args):
    """도구를 실제로 실행하는 것은 모델이 아니라 이 코드다.

    모델이 만든 인자는 신뢰하지 않는 입력으로 취급하여 검증한다.
    실행 오류는 프로그램 중단 대신 문자열로 돌려주어 모델이 대처하게 한다.
    """
    if name == "vat_calc":
        try:
            amount = float(args.get("amount", None))
        except (TypeError, ValueError):
            return "오류: amount는 숫자여야 합니다"
        if amount < 0:
            return "오류: 금액은 0 이상이어야 합니다"
        vat = round(amount * 0.1)
        return f"공급가액 {amount:,.0f}원, 부가세 {vat:,}원, 합계 {amount + vat:,.0f}원"
    return f"오류: 알 수 없는 도구 {name}"


def ask(question):
    """도구 호출 사이클을 단계별로 출력하며 질문 1건을 처리한다.

    한 응답의 tool_use 블록을 모두 실행해 tool_result를 같은 개수만큼 돌려보낸다. 반복은 MAX_ROUNDS까지.
    """
    messages = [{"role": "user", "content": question}]
    response = get_client().messages.create(
        model=MODEL, max_tokens=500, tools=[VAT_TOOL], messages=messages
    )
    print(f"[1차 응답] stop_reason = {response.stop_reason}")

    rounds = 0
    while response.stop_reason == "tool_use" and rounds < MAX_ROUNDS:
        rounds += 1
        results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            print(f"[호출 결정] 모델이 요청한 도구 = {block.name}, 인자 = {block.input}")
            result = run_tool(block.name, block.input)
            print(f"[코드가 실행] 결과 = {result}")
            results.append({"type": "tool_result", "tool_use_id": block.id, "content": result})
        messages.append({"role": "assistant", "content": response.content})
        messages.append({"role": "user", "content": results})
        response = get_client().messages.create(
            model=MODEL, max_tokens=500, tools=[VAT_TOOL], messages=messages
        )
        print(f"[결과 재주입 후 응답] stop_reason = {response.stop_reason}")

    final = "".join(b.text for b in response.content if b.type == "text")
    print(f"\n[최종 답변] {final}")


def main():
    if len(sys.argv) != 2:
        print(__doc__)
        sys.exit(1)
    ask(sys.argv[1])


if __name__ == "__main__":
    main()

# -*- coding: utf-8 -*-
"""임베딩 유사도 검색기 (week07 키트 · 완성되어 있음).

앞 주제 「API 연동과 RAG 챗봇」에서 두 함수(cosine_similarity · search)를 채워 만든 검색기와
같은 구조의 완성본입니다. 이 수업에서는 고치지 않고 도구의 부품으로 씁니다.

사용법:
    .venv/bin/python embed_search.py index 학과FAQ.md           # 문서를 청킹 · 임베딩하여 index.json 생성
    .venv/bin/python embed_search.py search "질문 문장"          # 질문과 가까운 상위 3개 청크 출력
    .venv/bin/python embed_search.py sim "문장1" "문장2"        # 두 문장의 유사도 출력

두 번째 문서(사내규정/복무규정.md)는 그 폴더에서 ../ 로 부릅니다.
    cd 사내규정 && ../.venv/bin/python ../embed_search.py index 복무규정.md

임베딩은 서버의 Ollama(공개 임베딩 모델)를 씁니다.
    - 환경변수 EMBED_MODEL: 임베딩 모델 이름 (과목 공지 값, 없으면 기본 nomic-embed-text)
    - 환경변수 CHUNK_MODE: "section"(제목 단위, 기본) 또는 "size"(고정 길이)
    - 환경변수 EMBED_BACKEND: "ollama"(기본) 또는 "fallback"
      임베딩 서버에 연결할 수 없으면 자동으로 예비 임베딩(fallback)으로 넘어갑니다.
      예비 임베딩은 설치가 필요 없는 저품질 대체 경로입니다. 검색은 되지만
      유사도 값은 참고용입니다. 색인과 질문은 같은 경로로 만들어야 합니다.

index.json은 명령을 실행한 폴더에 만들어지고 읽힙니다.
"""
import csv
import json
import math
import os
import sys
import urllib.error
import urllib.request

OLLAMA_URL = os.environ.get("OLLAMA_URL", "http://localhost:11434")
EMBED_MODEL = os.environ.get("EMBED_MODEL", "nomic-embed-text")
CHUNK_MODE = os.environ.get("CHUNK_MODE", "section")
EMBED_BACKEND = os.environ.get("EMBED_BACKEND", "ollama")  # "ollama" 또는 "fallback"
INDEX_FILE = "index.json"
TOP_K = 3
_FALLBACK_DIM = 1024
_notified = {"fallback": False}


def _embed_fallback(text):
    """예비 임베딩: 글자 두 개씩 묶은 조각(bigram)을 해시해 고정 길이 벡터로 만든다.
    의미를 이해하지는 못하지만 표현이 겹치는 정도를 잰다. 서버가 없을 때만 쓴다."""
    text = " ".join(text.lower().split())
    vec = [0.0] * _FALLBACK_DIM
    grams = [text[i:i + 2] for i in range(len(text) - 1)] or [text]
    for g in grams:
        h = 0
        for ch in g:
            h = (h * 131 + ord(ch)) % 1000003
        vec[h % _FALLBACK_DIM] += 1.0
    norm = math.sqrt(sum(v * v for v in vec)) or 1.0
    return [v / norm for v in vec]


def _embed_ollama(text):
    """Ollama 임베딩 호출. 현재 주소(/api/embed)를 먼저 쓰고, 옛 판 서버면 /api/embeddings로 다시 부른다."""
    calls = (
        ("/api/embed", {"model": EMBED_MODEL, "input": text}, lambda data: data["embeddings"][0]),
        ("/api/embeddings", {"model": EMBED_MODEL, "prompt": text}, lambda data: data["embedding"]),
    )
    for path, body, pick in calls:
        request = urllib.request.Request(
            OLLAMA_URL + path,
            data=json.dumps(body).encode("utf-8"),
            headers={"Content-Type": "application/json"},
        )
        try:
            with urllib.request.urlopen(request, timeout=30) as response:
                return pick(json.loads(response.read()))
        except urllib.error.HTTPError as e:
            if e.code == 404 and path == "/api/embed":
                continue  # 옛 판 서버: 다음 주소로
            raise


def embed(text):
    """텍스트 하나를 임베딩 벡터(숫자 목록)로 바꾼다. (완성되어 있음)"""
    global EMBED_BACKEND
    if EMBED_BACKEND == "ollama":
        try:
            return _embed_ollama(text)

def chunk(text):
    """문서를 청크 목록으로 자른다. (완성되어 있음, 기준은 CHUNK_MODE로 선택)"""
    if CHUNK_MODE == "section":
        parts = []
        current = []
        for line in text.split("\n"):
            if line.startswith("## ") and current:
                parts.append("\n".join(current).strip())
                current = []
            current.append(line)
        if current:
            parts.append("\n".join(current).strip())
        return [p for p in parts if p]
    if CHUNK_MODE == "size":
        size = 150
        return [text[i:i + size] for i in range(0, len(text), size)]
    raise ValueError("CHUNK_MODE는 section 또는 size")


def cosine_similarity(a, b):
    """두 벡터의 코사인 유사도를 반환한다. [실습에서 완성]

    공식: (a와 b의 내적) / (a의 크기 * b의 크기)
    내적은 같은 위치 성분끼리 곱해 모두 더한 값,
    크기는 각 성분 제곱합의 제곱근(math.sqrt)이다.
    """
    dot = sum(x * y for x, y in zip(a, b))  # zip: 두 목록을 같은 위치끼리 짝지음
    norm_a = math.sqrt(sum(x * x for x in a))
    norm_b = math.sqrt(sum(x * x for x in b))
    return dot / (norm_a * norm_b)


def search(query, index):
    """질문과 가까운 순으로 (유사도, 청크) 상위 TOP_K개를 반환한다. (완성되어 있음)

    순서: ① 질문을 embed로 벡터화 ② index의 모든 청크 벡터와
    cosine_similarity 계산 ③ 유사도 내림차순 정렬 후 앞 TOP_K개 반환.
    index는 [{"text": 청크, "vector": 벡터}, ...] 형태다.
    """
    query_vector = embed(query)
    scored = [(cosine_similarity(query_vector, item["vector"]), item["text"])
              for item in index]
    scored.sort(key=lambda pair: -pair[0])  # lambda는 이름 없이 한 줄로 쓰는 임시 함수. 점수가 큰 순서가 되도록 음수로 뒤집어 정렬
    return scored[:TOP_K]

전제: 환경변수 ANTHROPIC_API_KEY. COURSE_MODEL은 없으면 기본값을 씁니다.
"""

import sys

import anthropic

MODEL = os.environ.get("COURSE_MODEL", "claude-haiku-4-5-20251001")
MAX_ROUNDS = 5  # 도구 호출 반복의 상한 (무한 반복을 막는다)

_client = None


def get_client():
    """API 클라이언트를 처음 쓸 때 만든다(키는 환경변수 ANTHROPIC_API_KEY에서 읽는다)."""

    if _client is None:

# 도구 정의: 이름(영문), 설명, 입력 스키마. 모델은 이 정의만 보고 호출 여부를 판단한다.
VAT_TOOL = {
    "name": "vat_calc",
    "description": "공급가액(원)을 받아 부가세 10%와 합계 금액을 계산한다. "
                   "금액 계산이 필요한 질문에 사용한다. "
                   "부가세 제도나 세법을 설명하는 질문에는 사용하지 않는다.",
    "input_schema": {
        "type": "object",
        "properties": {
            "amount": {"type": "number", "description": "공급가액(원, 부가세 제외 금액)"}
        },
        "required": ["amount"],
    },
}


def run_tool(name, args):
    """도구 실행부. 실습 2에서 검증과 폴백을 붙인 판이다.

    모델이 만든 인자는 신뢰하지 않는 입력으로 취급하여 검증한다.
    실행 오류는 프로그램 중단 대신 문자열로 돌려주어 모델이 대처하게 한다.
    """
    if name == "vat_calc":
        try:
            amount = float(args.get("amount", None))
        except (TypeError, ValueError):
            return "오류: amount는 숫자여야 합니다"
        if amount < 0:
            return "오류: 금액은 0 이상이어야 합니다"
        vat = round(amount * 0.1)
        return f"공급가액 {amount:,.0f}원, 부가세 {vat:,}원, 합계 {amount + vat:,.0f}원"
    return f"오류: 알 수 없는 도구 {name}"


def ask(question):
    """도구 호출 사이클을 단계별로 출력하며 질문 1건을 처리한다.

    response = get_client().messages.create(
        model=MODEL, max_tokens=500, tools=[VAT_TOOL], messages=messages

        rounds += 1
        results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            print(f"[호출 결정] 모델이 요청한 도구 = {block.name}, 인자 = {block.input}")
            result = run_tool(block.name, block.input)
            print(f"[코드가 실행] 결과 = {result}")
            results.append({"type": "tool_result", "tool_use_id": block.id, "content": result})
        messages.append({"role": "assistant", "content": response.content})
        messages.append({"role": "user", "content": results})
        response = get_client().messages.create(
            model=MODEL, max_tokens=500, tools=[VAT_TOOL], messages=messages
        )
        print(f"[결과 재주입 후 응답] stop_reason = {response.stop_reason}")

    final = "".join(b.text for b in response.content if b.type == "text")
    print(f"\n[최종 답변] {final}")


def main():
    if len(sys.argv) != 2:
        print(__doc__)
        sys.exit(1)
    ask(sys.argv[1])


if __name__ == "__main__":
    main()

## API 키를 잃어버렸어요

API 키는 재발급할 수 있습니다. 교수에게 계정 이름과 함께 재발급을 요청하면 기존 키는 회수되고 새 키가 발급됩니다. 새 키는 홈 폴더의 환경 설정 파일에 다시 등록해야 합니다.

## 결석하면 실습 점수는 어떻게 되나요

제출물은 수업 참여를 전제로 하므로 결석 시 해당 회차 참여 점수가 빠집니다. 병결 등 증빙이 있는 결석은 다음 수업 전까지 보충 제출을 인정합니다. 보충 제출 방법은 교수에게 개별 문의합니다.

# 스킬 안티패턴 진단 기준

참고 서적 《하네스 엔지니어링 with 클로드 코드》의 안티패턴 3종을 수업용으로 축소 각색한 기준입니다. 원저 기준(본문 수백 줄)은 수업 재료 크기에 맞게 줄였습니다.

| 안티패턴 | 무엇이 문제인가 | 수업용 판정 기준 | 해결 방향 |
|---|---|---|---|
| ① 거대 본문 | 모든 지식을 SKILL.md 한 파일에 몰아넣어, 호출될 때마다 통째로 컨텍스트에 실린다 | 파일 60줄 초과 | 자주 쓰는 핵심만 본문에 남기고 상세 자료는 references/ 파일로 분리해 필요할 때만 읽게 한다 |
| ② 참조 분리 부재 | 서로 다른 세부 절차(문서 유형별, 상황별 분기)가 본문에 병렬로 나열되어 있다 | 상세 분기 4가지 이상인데 references/ 폴더가 없다 | 분기마다 references/ 파일 하나로 분리하고 본문에는 분기 표만 남긴다 |
| ③ 이유 없는 규칙 | 반드시·절대·금지 명령만 나열되어, 모델이 규칙의 취지를 모른 채 기계적으로 따르다 예외 상황에서 어긋난다 | 반드시/절대/금지가 5회 이상이고 왜냐하면/때문이 0회 | 각 규칙에 이유를 붙인다 (왜냐하면 ~이기 때문이다) |

한 SKILL.md가 둘 이상의 안티패턴에 동시에 걸리는 것은 흔한 일입니다. 샘플1은 ①과 ②에 함께 걸리며, 이때는 ①(분량)부터 해소하면 ②도 함께 풀립니다.

## 진단 절차

수치 신호는 명령으로 실측하고, 판정과 권고는 근거와 함께 기록합니다.

```bash
wc -l SKILL.md
grep -c -E "반드시|절대|금지" SKILL.md
grep -c -E "왜냐하면|때문" SKILL.md
ls references/ 2>/dev/null || echo "references 없음"
```

# 복무 규정 (발췌 · 실습용 가상 문서)

RAG 실습의 두 번째 검색 대상 문서입니다. 실제 회사의 규정이 아니라 실습용으로 작성한 가상 회사의 복무 규정에서 조항 일부를 뽑은 것입니다. 이 문서는 2026년 3월 1일 개정판이며, 조항 번호는 원문 번호를 그대로 두었습니다.

## 제3조 (근무시간)

근무시간은 오전 9시부터 오후 6시까지이며, 점심시간은 12:00부터 13:00까지입니다. 1주 근무시간은 40시간을 기준으로 합니다. 업무상 필요로 근무시간을 바꿀 때는 부서장이 미리 공지합니다.

## 제7조 (유연근무)

직원은 오전 8시부터 10시 사이에서 출근 시각을 골라 하루 8시간을 근무할 수 있습니다. 오전 10시부터 오후 4시까지의 코어타임에는 모든 직원이 근무해야 합니다. 유연근무는 적용 시작일 한 달 전까지 인사 시스템에서 신청합니다.

## 제12조 (지각과 조퇴)

출근 시각을 넘겨 도착하면 지각으로 처리합니다. 지각한 날에는 당일 안에 사유를 인사 시스템에 입력합니다. 조퇴는 부서장의 사전 승인을 받아야 하며, 긴급한 경우에는 사후 승인을 받습니다.

## 제15조 (연차휴가)

1년간 80% 이상 출근한 직원에게 15일의 연차휴가를 줍니다. 3년 이상 근속한 직원에게는 2년마다 1일을 더하며, 연차휴가는 최대 25일입니다. 연차휴가는 사용 3일 전까지 인사 시스템에서 신청하고, 반나절 단위(반차)로도 쓸 수 있습니다.

## 제16조 (경조사 휴가)

본인 결혼은 5일, 자녀 결혼은 1일, 부모와 배우자의 부모 사망은 5일, 조부모 사망은 3일의 경조사 휴가를 줍니다. 경조사 휴가는 연차휴가에서 차감하지 않습니다. 증빙 서류는 휴가가 끝난 뒤 7일 안에 제출합니다.

## 제18조 (병가)

질병이나 부상으로 근무할 수 없는 직원은 연간 10일까지 유급 병가를 쓸 수 있습니다. 병가가 연속 3일을 넘으면 의사의 진단서를 제출해야 합니다. 병가는 당일 오전 10시 전까지 부서장에게 알립니다.

## 제21조 (출장)

출장을 가려면 출발 전에 출장 신청서를 작성해 부서장의 승인을 받습니다. 국내 출장의 숙박비는 1박 10만 원을 상한으로 실비를 지급합니다. 출장이 끝나면 5일 안에 결과 보고서를 제출합니다.

## 제24조 (재택근무)

재택근무는 부서장 승인 후 인사 시스템에서 신청합니다. 재택근무는 주 2일 이내로 할 수 있습니다. 재택근무 중에는 회사가 지급한 장비로 VPN에 접속해 업무를 봅니다.

# 수정 전에 계획부터 검토한다 (Plan 모드)

근거: https://code.claude.com/docs/en/how-claude-code-works

- 상황: 파일을 어떻게 고칠지 모르는 큰 요청을 맡길 때, 바로 수정이 시작되면 검토할 기회가 없다.
- 방법: Shift+Tab을 눌러 권한 모드를 Plan으로 바꾼 뒤 요청한다. 계획만 출력되고 파일은 수정되지 않는다. 계획을 읽고 승인하면 실행된다.
- 예시: Plan 모드에서 `이 폴더의 테스트 구조를 pytest 표준으로 바꾸는 계획을 세워라`
- 주의점: Plan 모드를 켠 채로 실행까지 기대하면 아무 파일도 바뀌지 않는다. 계획 승인 후 모드가 전환되는지 확인한다.

# 작업이 바뀌면 컨텍스트를 정리한다 (/clear, /compact)

근거: https://code.claude.com/docs/en/how-claude-code-works

- 상황: 앞 작업의 대화 이력이 새 작업에 소음이 되고, 컨텍스트가 차면 오래된 내용이 유실된다.
- 방법: 이력을 버릴 때는 /clear, 요약해 이어갈 때는 /compact를 쓴다. 남길 내용은 /compact 뒤에 지시로 적는다.
- 예시: `/compact 테스트 실패 원인과 수정 방침은 반드시 남겨라`
- 주의점: /clear는 복구되지 않는다. 기록이 필요하면 /export로 먼저 내보낸다.

# 잘못된 수정은 체크포인트로 되돌린다 (/rewind)

근거: https://code.claude.com/docs/en/how-claude-code-works

- 상황: AI의 파일 수정이 의도와 다르게 흘러갔고, 손으로 되돌리기에는 변경이 많다.
- 방법: 입력창이 빈 상태에서 Esc를 두 번 누르거나 /rewind를 입력해 수정 전 체크포인트로 복원한다.
- 예시: 수정 직후 `/rewind` 입력, 목록에서 되돌릴 시점 선택
- 주의점: 체크포인트는 파일 변경만 복원한다. 외부 시스템에 반영된 작업(커밋 푸시, 배포)은 되돌리지 못한다.
